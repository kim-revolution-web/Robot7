from launch import LaunchDescription   # launch 파일 전체 구성을 담는 객체
from launch.actions import SetEnvironmentVariable, IncludeLaunchDescription  # 환경변수 설정, 다른 launch 파일 포함
from launch.launch_description_sources import PythonLaunchDescriptionSource   # 파이썬 launch 파일을 불러올 때 사용
from launch_ros.actions import Node    # ROS 2 노드를 실행할 때 사용

from ament_index_python.packages import get_package_share_directory  # 패키지의 share 폴더 경로 찾기
import os   # 경로 합치기 등에 사용하는 파이썬 기본 모듈


def generate_launch_description():
    # turtlebot3_bringup 패키지의 share 디렉토리 경로를 가져옴
    turtlebot3_bringup_dir = get_package_share_directory('turtlebot3_bringup')

    # turtlebot3_bringup 패키지 안의 launch/robot.launch.py 파일 경로를 만듦
    robot_launch = os.path.join(
        turtlebot3_bringup_dir,
        'launch',
        'robot.launch.py'
    )

    # launch에 실행할 내용들을 담아서 반환
    return LaunchDescription([
        # 터틀봇3 모델명을 burger로 환경변수에 설정
        # turtlebot3 관련 launch 파일들은 이 값을 보고 burger / waffle 등을 구분함
        SetEnvironmentVariable('TURTLEBOT3_MODEL', 'burger'),

        # robot.launch.py를 현재 launch 파일 안에 포함해서 같이 실행
        IncludeLaunchDescription(
            PythonLaunchDescriptionSource(robot_launch)
        ),

        # usb_cam 패키지의 usb_cam_node_exe 노드를 실행
        Node(
            package='usb_cam',              # 실행할 노드가 들어있는 패키지 이름
            executable='usb_cam_node_exe', # 실제 실행 파일 이름
            name='usb_cam',                # 실행될 노드 이름
            output='screen'                # 로그를 터미널 화면에 출력
        ),
    ])
